cmake_minimum_required(VERSION 3.18)
project(shadowfetch-drkonqi-pickup VERSION 4.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
option(ENABLE_BEHAVIOR_TESTS "Build original comparator and private journal test fixture" OFF)
find_package(Qt6 REQUIRED COMPONENTS Core Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd)
find_program(PATCH_COMMAND patch REQUIRED)

# All source inputs are shipped in this Debian source package. Configure never
# fetches from the network; reject a changed upstream archive before extraction.
set(archive "${CMAKE_CURRENT_SOURCE_DIR}/vendor/drkonqi-6.6.5.tar.xz")
file(SHA256 "${archive}" archive_sha256)
if(NOT archive_sha256 STREQUAL "b2b104d5f30224bca88280246ca990ac314d81cb1bf0d3857613ff683db92d82")
    message(FATAL_ERROR "Pinned KDE DrKonqi source archive SHA256 mismatch")
endif()
set(pristine "${CMAKE_CURRENT_BINARY_DIR}/upstream/drkonqi-6.6.5")
file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/upstream")
set(patched "${CMAKE_CURRENT_BINARY_DIR}/patched/src/coredump")
file(MAKE_DIRECTORY "${patched}/processor")
foreach(source IN ITEMS coredump.cpp coredump.h coredumpwatcher.cpp coredumpwatcher.h memory.h socket.h processor/main.cpp)
    configure_file("${pristine}/src/coredump/${source}" "${patched}/${source}" COPYONLY)
endforeach()
execute_process(COMMAND "${PATCH_COMMAND}" --batch --forward -p1 -i "${CMAKE_CURRENT_SOURCE_DIR}/patches/pickup-drains-journal.patch"
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/patched" RESULT_VARIABLE patch_result)
if(NOT patch_result EQUAL 0)
    message(FATAL_ERROR "Could not apply reviewed pickup lifecycle patch")
endif()

function(processor_target target base)
    add_executable(${target} "${base}/processor/main.cpp" "${base}/coredump.cpp" "${base}/coredumpwatcher.cpp")
    target_include_directories(${target} PRIVATE "${base}")
    target_link_libraries(${target} PRIVATE Qt6::Core Qt6::Network PkgConfig::SYSTEMD)
endfunction()
processor_target(shadowfetch-drkonqi-pickup "${patched}")
install(TARGETS shadowfetch-drkonqi-pickup DESTINATION libexec)
install(FILES systemd/10-shadowfetch-pickup.conf DESTINATION lib/systemd/user/drkonqi-coredump-pickup.service.d)

if(ENABLE_BEHAVIOR_TESTS)
    processor_target(processor-original "${pristine}/src/coredump")
    add_library(journal-fixture SHARED tests/journal_fixture.cpp)
    target_link_libraries(journal-fixture PRIVATE PkgConfig::SYSTEMD)
    set_target_properties(journal-fixture PROPERTIES AUTOMOC OFF)
endif()
